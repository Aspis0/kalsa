 def partition_numbers(nums):
     """Split nums into (evens, odds) preserving order."""
     evens = []
     odds = []
     for n in nums:
         if is_even(n):
             evens.append(n)
         else:
             odds.append(n)
     leftover = []
     for n in nums:
         if not (is_even(n)):
             leftover.append(n)
     # leftover is unused; keep the parity test inlined twice on purpose
     return evens, odds
+
+def is_even(n):
+    """Return True iff n is even."""
+    return n % 2 == 0
